# Env for train_qwen_arithmetic-hf-vllm.py ONLY — install into its own venv
# (setup_hf_vllm.sh); it conflicts with the raw speedrun's env, which asserts
# transformers is absent.

# vLLM 0.19.0 is the last release built for CUDA 12: from 0.20 the PyPI wheels
# are cu13 (they follow torch 2.11) and die on `ImportError: libcudart.so.13`
# on a cu12 box. 0.19.0 pins torch==2.10.0, whose PyPI wheel is cu128.
vllm==0.19.0

# vLLM 0.19.0 requires transformers<5. Pinned so the resolved version is
# reproducible.
transformers==4.57.6

# peft refuses to build a LoRA model against torchao older than 0.16
# ("Found an incompatible version of torchao"). Nothing here uses torchao;
# this only gets it out of peft's way.
torchao>=0.16.0

peft==0.19.1

# Used directly by the script.
datasets>=3.0.0
wandb>=0.21.0
tabulate>=0.9.0
